package main

import (
	"fmt"
	"math"
)

func main() {
	r := rectangle{length: 2, width: 3}
	c := circle{radius: 5}

	measure(r)
	measure(c)
	fmt.Println(c.diameter())

	// because it accepts unknown count of interfaces, it means it acceps any value and prints them.
	myPrinter(1, "fff", 12.3, r)

	printType(1)
	printType(false)
	printType(r)
}

// Geometry is an interface with two methods area and perimeter, both returning float 64 values.

type geometry interface {
	area() float64
	perim() float64
}

type rectangle struct {
	length, width float64
}

type circle struct {
	radius float64
}

// all we are doing is we are defining the internal code of this are method of the interface.
func (r rectangle) area() float64 {
	return r.length * r.width
}

func (c circle) area() float64 {
	return math.Pi * c.radius * c.radius
}

func (r rectangle) perim() float64 {
	return 2 * (r.length + r.width)
}

func (c circle) perim() float64 {
	return 2 * math.Pi * c.radius
}

// circle struct has one additioanl function
func (c circle) diameter() float64 {
	return 2 * c.radius
}

// because rectangle and circle they have both implemented the functions area and param from the interface geometry.
// Rect and circle can be used as geometry type
// anywhere where geometry type is accepted, Circle and rect would also be accepted over there,
// even though circle has an additional function.
// All a struct needs to do is implement all the methods defined in that interface.

func measure(g geometry) {
	fmt.Println(g)
	fmt.Println(g.area())
	fmt.Println(g.perim())
}

// That means I am ready to accept any number of parameters for my printer function.
// And when I use interface, that means they can be of any type.
// So because I used a variadic parameter that means we can accept any number of values of different types.
func myPrinter(i ...interface{}) {
	for _, v := range i {
		fmt.Println(v)
	}
}

// accepts only one value but with any type (number, string, bool, ...)
func myPrinter1(i interface{}) {
	fmt.Println(i)
}

func printType(i interface{}) {
	switch i.(type) {
	case int:
		fmt.Println("Int")
	case string:
		fmt.Println("String")
	case bool:
		fmt.Println("Bool")
	default:
		fmt.Println("Other")

	}
}
