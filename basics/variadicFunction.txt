package main

import "fmt"

func main() {
	// ... ellipsis
	// func functionName(param1 type1, param2 type2, param3 ...type3)returnType{}

	result := sum(1, 2, 3, 4, 5)
	fmt.Println("sum is:", result)

	statement, total := sum2("sum of numbers is:", 1, 2, 3)
	// statement, total := sum2("with no variadic param")
	fmt.Println(statement, total)

	// add slice as a variadic param
	numbers := []int{1, 2, 3, 7, 8, 9}
	result2 := sum(numbers...)
	fmt.Println("sum of slice is:", result2)

}

func sum(nums ...int) int {
	total := 0
	for _, v := range nums {
		total += v
	}
	return total
}

func sum2(returnString string, nums ...int) (string, int) {
	total := 0
	for _, v := range nums {
		total += v
	}
	return returnString, total
}
