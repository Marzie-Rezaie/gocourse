package main

import (
	"errors"
	"fmt"
)

func main() {

	// func functionName(parameters list) returnType{
	// 	codes
	// 	return value
	// }

	c := add(2, 2)
	fmt.Println(c)

	// Anonymos function (2 ways)

	// func() {
	// 	fmt.Println("Hello")
	// }()

	greet := func() {
		fmt.Println("Hello")
	}
	greet()

	// Function as type:
	Operation := add
	sum := Operation(1, 2)
	fmt.Println(sum)

	result1 := applyOperation(3, 4, add)
	fmt.Println("Result 1 is:", result1)

	result2 := createMultiplier(2)
	fmt.Println("Result2 is", result2(6))

	quotient, remainder := divide(4, 1)
	fmt.Printf("quotient is %d and remainder is %d\n", quotient, remainder)

	result, err := compare(3, 3)
	if err != nil {
		fmt.Println("error message is:", err)
	} else {
		fmt.Println("result is:", result)
	}

}

func add(a, b int) int {
	return a + b
}

// function that takes a function as an argument
func applyOperation(a, b int, operation func(int, int) int) int {
	return operation(a, b)
}

// function that returns a function
func createMultiplier(factor int) func(int) int {
	return func(x int) int {
		return x * factor
	}
}

// multiple return value function
func divide(a, b int) (int, int) {
	c := a / b
	d := a % b
	return c, d
}

// fuction with error return
func compare(a, b int) (string, error) {
	if a > b {
		return "a is greater", nil
	} else if b > a {
		return "b is greater", nil
	} else {
		return "", errors.New("Unable to compare")
	}
}

// named return values
func divide2(a, b int) (quotient int, remainder int) {
	quotient = a / b
	remainder = a % b
	// there is no need to mention again in the return
	return
}
